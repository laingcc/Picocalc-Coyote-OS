## Wi-Fi + lwIP (Pico W / Pico 2 W).  The lwip_poll arch keeps everything on
## the main loop; it must not be combined with pico_cyw43_arch_none.  Boards
## without a CYW43 radio still build, with networking compiled out.
if (TARGET pico_cyw43_arch_lwip_poll)
    # lwIP needs an lwipopts.h on the include path.  It is generated here so
    # the options live next to the libraries they configure.
    set(COYOTE_GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
    file(WRITE ${COYOTE_GENERATED_DIR}/lwipopts.h.in [=[
#ifndef COYOTE_LWIPOPTS_H
#define COYOTE_LWIPOPTS_H

/* Generated by CMakeLists.txt; edit it there. */

/* Bare-metal, raw API only: no sockets, no netconn, no OS. */
#define NO_SYS                      1
#define LWIP_SOCKET                 0
#define LWIP_NETCONN                0

/* lwIP allocates from its own fixed pools, never the C heap. */
#define MEM_LIBC_MALLOC             0
#define MEM_ALIGNMENT               4
#define MEM_SIZE                    8000
#define MEMP_NUM_TCP_PCB            2
#define MEMP_NUM_TCP_SEG            16
#define MEMP_NUM_ARP_QUEUE          8
/* DHCP, DNS (one per lookup in flight) and SNTP each hold a UDP PCB. */
#define MEMP_NUM_UDP_PCB            6
/* The SNTP client keeps up to two timers of its own. */
#define MEMP_NUM_SYS_TIMEOUT        (LWIP_NUM_SYS_TIMEOUT_INTERNAL + 2)
#define PBUF_POOL_SIZE              16

#define LWIP_ARP                    1
#define LWIP_ETHERNET               1
#define LWIP_ICMP                   1
#define LWIP_RAW                    0
#define LWIP_IPV4                   1
#define LWIP_TCP                    1
#define LWIP_UDP                    1
#define LWIP_DHCP                   1
#define LWIP_DNS                    1
/* Resolve *.local names (a common way to name the chat host) with one-shot
 * multicast DNS queries. */
#define LWIP_DNS_SUPPORT_MDNS_QUERIES 1

#define TCP_MSS                     1460
#define TCP_WND                     (4 * TCP_MSS)
#define TCP_SND_BUF                 (2 * TCP_MSS)
#define TCP_SND_QUEUELEN            ((4 * (TCP_SND_BUF) + (TCP_MSS - 1)) / (TCP_MSS))
#define LWIP_TCP_KEEPALIVE          1

#define LWIP_NETIF_STATUS_CALLBACK  1
#define LWIP_NETIF_LINK_CALLBACK    1
#define LWIP_NETIF_HOSTNAME         1
#define LWIP_NETIF_TX_SINGLE_PBUF   1
#define DHCP_DOES_ARP_CHECK         0
#define LWIP_DHCP_DOES_ACD_CHECK    0
#define LWIP_CHKSUM_ALGORITHM       3

/* SNTP client (ai/time_service.c): look the server up by name, ask as soon as
 * it is started and then hourly, and only accept an answer from the address
 * that was asked. */
#define SNTP_SERVER_DNS             1
#define SNTP_STARTUP_DELAY          0
#define SNTP_UPDATE_DELAY           3600000
#define SNTP_CHECK_RESPONSE         1
#ifndef __ASSEMBLER__
#include <stdint.h>
void time_service_on_sntp(uint32_t epoch);
#endif
#define SNTP_SET_SYSTEM_TIME(sec)   time_service_on_sntp(sec)

#define MEM_STATS                   0
#define SYS_STATS                   0
#define MEMP_STATS                  0
#define LINK_STATS                  0
#define LWIP_STATS                  0

#endif
]=])
    # Copy only when the content changes so reconfiguring does not rebuild lwIP.
    configure_file(${COYOTE_GENERATED_DIR}/lwipopts.h.in ${COYOTE_GENERATED_DIR}/lwipopts.h COPYONLY)
    target_include_directories(coyote PRIVATE ${COYOTE_GENERATED_DIR})
    target_compile_definitions(coyote PRIVATE COYOTE_HAS_WIFI=1)
    target_link_libraries(coyote PRIVATE pico_cyw43_arch_lwip_poll pico_lwip_sntp)
else()
    message(WARNING "PICO_BOARD has no CYW43 radio: building without Wi-Fi (use -DPICO_BOARD=pico2_w)")
endif()


pico_enable_stdio_usb(coyote 0)
pico_enable_stdio_uart(coyote 1)
